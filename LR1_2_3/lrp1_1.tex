\section*{Tools Used}

\end{itemize}

\section*{Process}

\subsection*{Code:}
\begin{minted}[breaklines, linenos]{gas}
DATA    SEGMENT
    CR EQU 0DH
    LF EQU 0AH
    MSG1 DB 'ENTER A LOWERCASE LETTER: $'
    MSG2 DB CR,LF, 'IN UPPER CASE IT IS: '
    CHAR DB ?, '$'
DATA    ENDS
CODE    SEGMENT
    ASSUME CS:CODE, DS:DATA
    
START:  
    MOV AX, DATA
    MOV DS, AX
          
    LEA DX, MSG1
    MOV AH, 9
    INT 21h
    MOV AH, 1
    INT 21h
    
    SUB AL, 20H
    MOV CHAR, AL
        
    LEA DX,MSG2
    MOV AH, 9
    INT 21h

    MOV AH, 4CH
    INT 21h    
     
CODE    ENDS

END START
\end{minted}
